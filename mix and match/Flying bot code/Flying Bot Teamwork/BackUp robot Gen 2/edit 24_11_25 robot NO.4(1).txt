// generating and setting random seed
void initializeRandomSeed()
{
  wait(100, msec);
  double xAxis = BrainInertial.acceleration(xaxis) * 1000;
  double yAxis = BrainInertial.acceleration(yaxis) * 1000;
  double zAxis = BrainInertial.acceleration(zaxis) * 1000;
  // Combine these values into a single integer
  int seed = int(
      xAxis + yAxis + zAxis);
  // Set the seed
  srand(seed);
}

void vexcodeInit()
{

  // Initializing random seed.
  initializeRandomSeed();
}

// User defined function
void Place_beam()
{
  Motor3Left.stop();
  Motor9Right.stop();
  front_back = true;
  MotorBeam.setVelocity(65.0, percent);
  guide.retract(cylinder1);
  MotorBeam.spinFor(forward, 115.0, degrees, true);
  Pneumatic_Pin_Beam.retract(cylinder2);
  guide.extend(cylinder1);
  Motor9Right.stop();
  Motor3Left.stop();
  Motor3Left.setVelocity(85.0, percent);
  Motor9Right.setVelocity(85.0, percent);
  stop = true;
  MotorBeam.spinFor(reverse, 50.0, degrees, true);
  Motor3Left.spinFor(forward, 130.0, degrees, false);
  Motor9Right.spinFor(forward, 130.0, degrees, false);
  wait(0.4, seconds);
  stop = false;
  Drop_down_beam();
  // MotorBeam.spinFor(forward, 185.0, degrees, true);
  // MotorBeam.setPosition(0.0, degrees);
  isBusy = false;
}

// User defined function
void Drop_down_beam()
{
  stop = false;
  guide.extend(cylinder1);
  wait(100, msec);
  MotorBeam.setStopping(hold);
  beam ? MotorBeam.setVelocity(100, percent) : MotorBeam.setVelocity(45, percent);
  MotorBeam.spin(forward);
  MotorBeam.setMaxTorque(100, percent);
  wait(100, msec);
  while (MotorBeam.velocity(vex::velocityUnits::pct) > 1)
  {
    MotorBeam.spin(forward);
    wait(1, msec);
  }
  // MotorBeam.spinFor(forward, 1000.0, degrees, false);
  // wait(1500, msec);
  MotorBeam.setStopping(brake);
  MotorBeam.setMaxTorque(0, percent);
  MotorBeam.stop();
  isBusy = false;
}

// User defined function
void Grab_then_up()
{
  stop = true;
  if (Motor3Left.velocity(vex::velocityUnits::pct) > 1)
  {
    for (float speedMotorPin = Motor3Left.velocity(vex::velocityUnits::pct); speedMotorPin > 1; speedMotorPin -= 5)
    {
      Motor3Left.setVelocity(speedMotorPin, percent);
      Motor9Right.setVelocity(speedMotorPin, percent);
      Motor3Left.spin(forward);
      Motor9Right.spin(forward);
      wait(10, msec);
    }
  }
  stop = false;
  MotorPin.setStopping(hold);
  Pneumatic_Pin_Beam.extend(cylinder1);
  MotorPin.spinFor(reverse, 160.0, degrees, true);
  guide.extend(cylinder2);
  isBusy = false;
}

// User defined function
void Drop_down()
{
  Motor3Left.setStopping(hold);
  Motor9Right.setStopping(hold);
  guide.retract(cylinder2);
  wait(0.35, seconds);
  MotorPin.setStopping(coast);
  MotorPin.spinFor(forward, 155.0, degrees, false);
  wait(0.3, seconds);
  Pneumatic_Pin_Beam.retract(cylinder1);
  MotorPin.stop();
  wait(0.3, seconds);
  Motor3Left.setStopping(coast);
  Motor9Right.setStopping(coast);
}

// User defined function
void Drop_down_Grab_Up()
{
  stop = true;
  if (Motor3Left.velocity(vex::velocityUnits::pct) > 15)
  {
    Motor3Left.spin(forward);
    Motor9Right.spin(forward);
    for (float speedMotorPin = Motor3Left.velocity(vex::velocityUnits::pct); speedMotorPin > 1; speedMotorPin -= 5)
    {
      Motor3Left.setVelocity(speedMotorPin, percent);
      Motor9Right.setVelocity(speedMotorPin, percent);
      Motor3Left.spin(forward);
      Motor9Right.spin(forward);
      wait(10, msec);
    }
  }
  MotorPin.setStopping(hold);
  Motor3Left.stop();
  Motor9Right.stop();
  wait(0.2, seconds);
  guide.retract(cylinder2);
  wait(0.2, seconds);
  MotorPin.spinFor(forward, 80.0, degrees, true);
  Pneumatic_Pin_Beam.retract(cylinder1);
  MotorPin.spinFor(forward, 80.0, degrees, true);
  // wait(0.2, seconds);
  Motor3Left.setVelocity(50.0, percent);
  Motor9Right.setVelocity(50.0, percent);
  Motor3Left.spinFor(forward, 140.0, degrees, false);
  Motor9Right.spinFor(forward, 140.0, degrees, false);
  wait(0.3, seconds);
  Pneumatic_Pin_Beam.extend(cylinder1);
  MotorPin.spinFor(reverse, 160.0, degrees, false);
  wait(0.25, seconds);
  guide.extend(cylinder2);
  stop = false;
  isBusy = false;
}

// "when Controller ButtonLUp pressed" hat block
void LUp_pressed()
{
  wait(0.1, seconds);
  if (!isBusy)
  {
    Lup_Init();
    isBusy = true;
    if (Lup == 1.0)
    {
      front_back = false;
      Grab_Beam_up();
      // Spin_Robot();
      Lup = 1.0;
      autobeam = true;
    }
    else if (Lup == 2.0)
    {
      stop = true;
      beam = true;
      Place_beam();
      Lup = 0.0;
      autobeam = false;
    }
    isBusy = false;
  }
}

// "when Controller ButtonLDown pressed" hat block
void LDown_pressed()
{
  Ldown_Init();
  wait(0.1, seconds);
  if (!isBusy && Lup == 1.0)
  {
    if (Ldown >= 1.0)
    {
      Drop_down_beam();
      Lup = 0.0;
      Ldown = 0.0;
      front_back = true;
      autobeam = false;
      stand_off_up = false;
    }
    isBusy = false;
  }
}
void Drive()
{
  if (!Controller.ButtonEUp.pressing())
  {
    if (front_back)
    {
      Motor9Right.setVelocity((((Controller.AxisA.position() * Controller.AxisA.position()) * (Controller.AxisA.position() * 0.0001) - (Controller.AxisB.position() * Controller.AxisB.position()) * (Controller.AxisB.position() * 0.0001)) * Kspeed), percent);
      Motor3Left.setVelocity((((Controller.AxisA.position() * Controller.AxisA.position()) * (Controller.AxisA.position() * 0.0001) + (Controller.AxisB.position() * Controller.AxisB.position()) * (Controller.AxisB.position() * 0.0001)) * Kspeed), percent);
      Motor3Left.spin(forward);
      Motor9Right.spin(forward);
    }
    else if (!front_back)
    {
      Motor9Right.setVelocity((((Controller.AxisA.position() * Controller.AxisA.position()) * (Controller.AxisA.position() * 0.0001) + (Controller.AxisB.position() * Controller.AxisB.position()) * (Controller.AxisB.position() * 0.0001)) * Kspeed), percent);
      Motor3Left.setVelocity((((Controller.AxisA.position() * Controller.AxisA.position()) * (Controller.AxisA.position() * 0.0001) - (Controller.AxisB.position() * Controller.AxisB.position()) * (Controller.AxisB.position() * 0.0001)) * Kspeed), percent);
      Motor3Left.spin(reverse);
      Motor9Right.spin(reverse);
    }
    if ((Controller.AxisA.position() > -5.0 and Controller.AxisA.position() < 5) and (Controller.AxisB.position() > -5.0 and Controller.AxisB.position() < 5))
    {
      Motor3Left.stop();
      Motor9Right.stop();
    }
  }
}

// "when started" hat block
int seting()
{
  MotorPin.setPosition(0.0, degrees);
  MotorBeam.setPosition(0.0, degrees);
  MotorPin.setVelocity(80.0, percent);
  MotorPin.setMaxTorque(100.0, percent);
  MotorBeam.setVelocity(100.0, percent);
  MotorBeam.setMaxTorque(100.0, percent);
  MotorPin.setStopping(hold);
  MotorBeam.setStopping(hold);
  Motor3Left.setStopping(coast);
  Motor9Right.setStopping(coast);
  guide.retract(cylinder1);
  guide.retract(cylinder2);
  Pneumatic_Pin_Beam.retract(cylinder1);
  Pneumatic_Pin_Beam.retract(cylinder2);
  Pneumatic_Pin_Beam.pumpOn();
  Brain.Screen.setCursor(1, 1);
  Brain.Screen.print("Flying Bot CNX");
  pin = true;
  phase = true;
  beam = true;
  isBusy = false;
  Rup = 0.0;
  Rdown = 0.0;
  Lup = 1.0;
  Ldown = 1.0;
  R3 = 0.0;
  front_back = true;
  boot = false;
  Kspeed = 1.0;
  stop = false;
  autobeam = false;
  stand_off_up = true;
  start_stand_off_up = true;
  Drop_down_beam();
  MotorBeam.spinFor(reverse, 510.0, degrees);
  while (true)
  {
    if (!(Lup == 3.0 || Rup == 2.0 || stop))
    {
      Drive();
    }
    wait(25, msec);
  }
  return 0;
}

// "when started" hat block
int LED_controller()
{
  while (true)
  {
    if (front_back)
    {
      TouchLED12.setColor(blue);
      if (blink)
      {
        TouchLED12.setColor(green);
        wait(500, msec);
      }
      else
      {
        if (!boot)
        {
          TouchLED12.setColor(blue);
          wait(0.02, seconds);
          TouchLED12.setColor(colorType::none);
        }
      }
    }
    else
    {
      TouchLED12.setColor(orange);
      if (blink)
      {
        TouchLED12.setColor(green);
        wait(500, msec);
      }
      else
      {
        if (Distance1Left.objectDistance(inches) <= 6.3)
        {
          TouchLED12.setColor(green);
          wait(0.02, seconds);
          TouchLED12.setColor(colorType::none);
        }

        else if (!boot)
        {
          TouchLED12.setColor(orange);
          wait(0.02, seconds);
          TouchLED12.setColor(colorType::none);
        }
      }
    }
    wait(20, msec);
  }
  return 0;
}

// "when Controller ButtonRUp pressed" hat block
void RUp_pressed()
{
  Rup_Init();
  if (!isBusy)
  {
    isBusy = true;
    if (Rup == 1.0)
    {
      Grab_then_up();
      pin = false;
    }
    else if (Rup == 2.0)
    {
      Drop_down_Grab_Up();
      Rup = 1.0;
      pin = false;
    }
    else
    {
      Rup = 1.0;
    }
    isBusy = false;
  }
}

// "when started" hat block
int Time()
{
  Brain.Timer.reset();
  Brain.Screen.setFont(mono20);
  while ((Brain.Timer.value() < 120.0))
  {
    Brain.Screen.setCursor(1, 1);
    wait(1, msec);
  }
  Brain.programStop();
  return 0;
}

  }
}

// "when Controller AxisD changed" hat block
void ControllerDChanged()
{
  if (Controller.AxisD.position() >= 90.0)
  {
    guide.retract(cylinder2);
  }
  else if (Controller.AxisD.position() <= -90.0)
  {
    guide.extend(cylinder2);
  }
}

// "when Controller AxisC changed" hat block
void ControllerCChanged()
{
  MotorBeam.setMaxTorque(100, percent);
  MotorBeam.setVelocity(100.0, percent);

  if (Controller.AxisC.position() > 90.0)
  {
    if (stand_off_up)
    {
      stop = true;
      Motor3Left.stop();
      Motor9Right.stop();
      front_back = true;
      if (start_stand_off_up)
      {
        MotorBeam.setVelocity(100.0, percent);
        MotorBeam.spinFor(forward, 110.00, degrees);
        Pneumatic_Pin_Beam.retract(cylinder2);
        Motor9Right.stop();
        Motor3Left.stop();
        MotorBeam.spinFor(reverse, 100.00, degrees);
        start_stand_off_up = false;
      }
      else
      {
        MotorBeam.setVelocity(45.0, percent);
        MotorBeam.spinFor(forward, 125.00, degrees);
        Pneumatic_Pin_Beam.retract(cylinder2);
        Motor9Right.stop();
        Motor3Left.stop();
        MotorBeam.spinFor(reverse, 70.00, degrees);
      }
      Motor3Left.setVelocity(100.0, percent);
      Motor9Right.setVelocity(100.0, percent);
      Motor3Left.spinFor(forward, 350.0, degrees, false);
      Motor9Right.spinFor(forward, 350.0, degrees, false);
      wait(0.6, seconds);
      beam = true;
      Drop_down_beam();
      stop = false;
      stand_off_up = false;
      Lup = 0.0;
    }
  }
  else if (Controller.AxisC.position() < -90.0)
  {
    if (!stand_off_up)
    {
        MotorBeam.spinFor(reverse, 520.0, degrees, false);
        stand_off_up = true;
        Lup = 1.0;
      
    }
  }
}

// "when Controller ButtonR3 pressed" hat block
void ControllerButtonR3_pressed()
{
  Brain.Timer.reset();
  MotorPin.setVelocity(100.0, percent);
  MotorPin.setMaxTorque(100.0, percent);
  if (!isBusy && Rup == 1.0)
  {
    isBusy = true;
    if (Rup == 1.0)
    {
      guide.retract(cylinder2);
      MotorPin.spinFor(reverse, 575.0, degrees, false);
      wait(1.5, seconds);
      Pneumatic_Pin_Beam.retract(cylinder1);
      // front_back = false;
      wait(0.4, seconds);
      MotorPin.spinFor(forward, 675.0, degrees);
      // wait(0.3, seconds);
      // Grab_Beam_up();
      MotorPin.setStopping(coast);
      MotorPin.stop();
      // Lup = 1.0;
      Rup = 0.0;
      autobeam = true;
      pin = true;
      wait(0.1, seconds);
    }
    isBusy = false;
  }
}

// "when started" hat block
int autoBeam()
{
  while (true)
  {
    while ((autobeam && Lup == 1.0 && !isBusy))
    {
      isBusy = true;
      if (Distance1Left.objectDistance(inches) <= 6.3)
      {
        stop = true;
        Motor3Left.setStopping(hold);
        Motor9Right.setStopping(hold);
        Motor3Left.stop();
        Motor9Right.stop();
        Place_beam();
        stop = false;
        Lup = 0.0;
        beam = true;
        autobeam = false;
        Motor3Left.setStopping(coast);
        Motor9Right.setStopping(coast);
      }
      isBusy = false;
      wait(300, msec);
    }
    wait(500, msec);
  }
  return 0;
}
